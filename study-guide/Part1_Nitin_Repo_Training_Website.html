def set_seed(seed):
    """Make a run repeatable: the same seed gives the same random numbers."""
    random.seed(seed)                          # Python's own random numbers
    np.random.seed(seed)                       # numpy's random numbers
    torch.manual_seed(seed)                    # PyTorch's random numbers (CPU and GPU)
    torch.backends.cudnn.deterministic = True  # the GPU uses the same convolution method every time
    torch.backends.cudnn.benchmark = False     # do not let the GPU try methods and pick by speed
    # note: two different laptops can still give slightly different numbers


def get_device():
    """Return the GPU if PyTorch can use one, otherwise the CPU."""
    return torch.device("cuda" if torch.cuda.is_available() else "cpu")

def build_model(model_name):
    """Return (model, learning rate) for one of our model names."""
    if model_name in SCRATCH:
        return models.build_scratch_cnn(**SCRATCH[model_name]), 1e-3

4.2.5 Bytes to numbers, and the augmentation (lines 90 to 106)

def _prepare(batch_bytes, augment):
    """Bytes (0 to 255) to normalised floats. With augment: random 90-degree turns and flips, per image."""
    x = batch_bytes.float() / 255.0
    if augment:
        n = x.shape[0]
        turns = torch.randint(0, 4, (n,), device=x.device)
        for k in (1, 2, 3):
            chosen = turns == k
            if chosen.any():
                x[chosen] = torch.rot90(x[chosen], k, dims=(2, 3))
        flip_sideways = torch.rand(n, device=x.device) < 0.5
        flip_upside_down = torch.rand(n, device=x.device) < 0.5
        x[flip_sideways] = x[flip_sideways].flip(3)
        x[flip_upside_down] = x[flip_upside_down].flip(2)
    mean = torch.tensor(IMAGENET_MEAN, device=x.device).view(1, 3, 1, 1)
    std = torch.tensor(IMAGENET_STD, device=x.device).view(1, 3, 1, 1)
    return (x - mean) / std

4.2.6 Scoring without training (lines 109 to 122)

def _score(model, rows, images, labels, loss_function):
    """Average loss, accuracy and predictions for some rows, without augmentation."""
    model.eval()
    total_loss = 0.0
    predictions = []
    with torch.no_grad():
        for start in range(0, len(rows), 256):
            batch = rows[start:start + 256]
            scores = model(_prepare(images[batch], augment=False))
            total_loss += loss_function(scores, labels[batch]).item() * len(batch)
            predictions.append(scores.argmax(dim=1))
    predictions = torch.cat(predictions)
    accuracy = (predictions == labels[rows]).float().mean().item()
    return total_loss / len(rows), accuracy, predictions

4.2.8 The train() function (lines 139 to 227)

def train(model_name, fraction, seed, epochs=MAX_EPOCHS, metrics_csv=METRICS_CSV, save_weights=None):
    """Train one model with the team settings and add ONE row to results/metrics.csv.

    model_name:   one of SCRATCH or TRANSFER above
    fraction:     share of the TRAIN split to use, from 0.05 to 1.0 (stratified). Validation and test never shrink.
    seed:         makes the run repeatable (starting weights, which images, their order, augmentation)
    save_weights: keep the best-epoch weights in models/ (not in git). Default: only when fraction is 1.0.

    The best epoch is chosen on the VALIDATION set. The test set is scored ONCE, after training.
    Returns a dict with the result and the training history.
    """
    if not 0 < fraction <= 1:
        raise ValueError("fraction must be above 0 and at most 1")
    device = get_device()
    set_seed(seed)

    images, labels, row_of_path = _load_all_images(device)
    train_rows = _rows("train", fraction, seed, row_of_path, device)
    val_rows = _rows("val", 1.0, seed, row_of_path, device)
    test_rows = _rows("test", 1.0, seed, row_of_path, device)

    model, learning_rate = build_model(model_name)
    model = model.to(device)
    loss_function = torch.nn.CrossEntropyLoss()
    # Only the layers that are allowed to learn go to the optimiser (matters for the frozen models).
    # foreach=False is the same setting Tejash used in his runs.
    optimizer = torch.optim.Adam([p for p in model.parameters() if p.requires_grad],
                                 lr=learning_rate, foreach=False)
    shuffler = torch.Generator().manual_seed(seed)  # decides the order of the images, so a run repeats

    history = {"train_loss": [], "train_accuracy": [], "val_loss": [], "val_accuracy": []}
    best_accuracy, best_epoch, best_weights = -1.0, 0, None
    started = time.perf_counter()

    for epoch in range(1, epochs + 1):
        epoch_started = time.perf_counter()
        model.train()
        total_loss = torch.zeros((), device=device)
        correct = torch.zeros((), device=device)
        seen = 0

        order = train_rows[torch.randperm(len(train_rows), generator=shuffler).to(device)]
        for start in range(0, len(order), BATCH_SIZE):
            batch = order[start:start + BATCH_SIZE]
            if len(batch) < 2:  # BatchNorm cannot work with a single image
                continue
            x = _prepare(images[batch], augment=True)
            y = labels[batch]
            optimizer.zero_grad()
            scores = model(x)
            loss = loss_function(scores, y)
            loss.backward()
            optimizer.step()
            total_loss += loss.detach() * len(batch)
            correct += (scores.argmax(dim=1) == y).sum()
            seen += len(batch)

        val_loss, val_accuracy, _ = _score(model, val_rows, images, labels, loss_function)
        history["train_loss"].append(total_loss.item() / seen)
        history["train_accuracy"].append(correct.item() / seen)
        history["val_loss"].append(val_loss)
        history["val_accuracy"].append(val_accuracy)
        print(f"{model_name} | fraction {fraction:g} | seed {seed} | epoch {epoch:02d}/{epochs} | "
              f"train loss {history['train_loss'][-1]:.4f}, acc {history['train_accuracy'][-1]:.3f} | "
              f"val loss {val_loss:.4f}, acc {val_accuracy:.3f} | {time.perf_counter() - epoch_started:.1f} s")

        if val_accuracy > best_accuracy:  # keep the best epoch, chosen on validation
            best_accuracy, best_epoch = val_accuracy, epoch
            best_weights = {name: value.detach().cpu().clone() for name, value in model.state_dict().items()}

    model.load_state_dict(best_weights)

    # The test set is used here, once, and nothing is decided from it.
    _, accuracy, predictions = _score(model, test_rows, images, labels, loss_function)
    macro_f1 = f1_score(labels[test_rows].cpu().numpy(), predictions.cpu().numpy(), average="macro")
    row = {"model": model_name, "fraction": fraction, "seed": seed, "accuracy": accuracy, "macro_f1": macro_f1}
    _save_row(metrics_csv, row)

    if save_weights is None:
        save_weights = fraction == 1.0
    if save_weights:
        MODELS_DIR.mkdir(exist_ok=True)
        torch.save(model.state_dict(), MODELS_DIR / f"{model_name}_frac{fraction:g}_seed{seed}.pth")

    seconds = time.perf_counter() - started
    print(f"TEST (best epoch {best_epoch}): accuracy {accuracy:.4f}, macro-F1 {macro_f1:.4f} "
          f"| {seconds / 60:.1f} min, {seconds / epochs:.1f} s per epoch")
    return {**row, "best_epoch": best_epoch, "val_accuracy": best_accuracy,
            "seconds_per_epoch": seconds / epochs, "history": history}

def make_small_cnn(torch):
    """Small scratch CNN: 3 blocks of Conv-BatchNorm-ReLU-MaxPool, then a Linear to 10 classes."""
    nn = torch.nn

        return nn.Sequential(nn.Conv2d(cin, cout, 3, padding=1), nn.BatchNorm2d(cout),

    return nn.Sequential(block(3, 32), block(32, 64), block(64, 128),
                         nn.AdaptiveAvgPool2d(1), nn.Flatten(), nn.Linear(128, 10))

def benchmark(torch, model, device, steps, warmup=5):
    """Train on random 64x64 data and return images per second."""
    model = model.to(device)
    model.train()
    opt = torch.optim.Adam(model.parameters())
    loss_fn = torch.nn.CrossEntropyLoss()
    x = torch.rand(BATCH, 3, 64, 64, device=device)
    y = torch.randint(0, 10, (BATCH,), device=device)

    def one_step():
        opt.zero_grad()
        loss_fn(model(x), y).backward()
        opt.step()

    for _ in range(warmup):
        one_step()
    sync(torch, device)
    start = time.perf_counter()
    for _ in range(steps):
        one_step()
    sync(torch, device)
    return steps * BATCH / (time.perf_counter() - start)

5. How it works end to end: train('mobilenet_v2_full', 1.0, 0)

Assume the person has run check_setup.py, the data is in data/eurosat, and the GPU is available.

  1. The call. train receives model_name="mobilenet_v2_full", fraction=1.0, seed=0. It uses the defaults: epochs=20, metrics_csv=results/metrics.csv, save_weights=None.
  2. Check the fraction. 0 < 1.0 <= 1 is True, so no error.
  3. Pick the device. get_device() returns "cuda".
  4. Set the seed. set_seed(0) seeds Python, numpy and PyTorch, and sets cuDNN to deterministic. From here on, every random step repeats.
  5. Load the images (first call in this session only). _load_all_images("cuda") reads splits.csv, opens all 27,000 pictures, stacks them into one uint8 block of shape (27000, 3, 64, 64), and puts it on the GPU with the labels and the path-to-row lookup. It prints the "about 20 seconds" message. Later calls in the same session return the cached block immediately.
  6. Choose the rows. _rows asks get_loader for the train split with fraction=1.0 (so all 18,900 rows; no subset), and for validation and test (4,050 each). It turns the paths into row numbers on the GPU.
  7. Build the model. build_model("mobilenet_v2_full") finds ("mobilenet_v2", False) in TRANSFER. Because freeze is False, it calls models.build_transfer_model("mobilenet_v2", False). That loads ImageNet weights (from the torch cache, or downloaded the first time), leaves every layer trainable, and replaces the 1000-class output with a new 10-class linear layer. The learning rate is 1e-4.
  8. Prepare the training. The model moves to the GPU. Loss is CrossEntropyLoss. Adam gets every parameter (all have requires_grad True) with lr=1e-4 and foreach=False. A seeded generator is made for the shuffle.
  9. Epoch 1. model.train(). The training rows are shuffled with the seeded generator. The rows are cut into 296 batches of 64 (the last one has 20 images). For each batch: _prepare (bytes to 0 to 1, random 90-degree turns and flips per image, normalisation), zero_grad, forward pass, loss, backward, step, and the running totals are updated on the GPU.
  10. Validate. _score puts the model in eval(), scores the 4,050 validation images in groups of 256 without gradients, and returns loss and accuracy. The four history numbers are stored, and one line is printed with the epoch number, losses, accuracies and seconds.
  11. Best so far? If the validation accuracy is above the best so far, the epoch number and a CPU copy of all weights are kept.
  12. Repeat steps 9 to 11 for epochs 2 to 20. Each epoch reshuffles the training rows, and the seeded generator makes the shuffles repeatable.
  13. Restore the best. After epoch 20, load_state_dict(best_weights) puts back the weights of the best validation epoch (not necessarily epoch 20).
  14. Test once. _score scores the 4,050 test images with those weights. Accuracy comes from the scores, and macro-F1 comes from sklearn.metrics.f1_score(..., average="macro") on the predicted and true labels.
  15. Write the row. A dictionary {model, fraction, seed, accuracy, macro_f1} is made. _save_row reads results/metrics.csv, removes any old row with the same model, fraction and seed, appends the new row, and writes the file with \n line endings.
  16. Save the weights. save_weights is None, and fraction == 1.0, so it is True. models/ is created if needed, and the best state dictionary is saved as models/mobilenet_v2_full_frac1_seed0.pth. The folder is ignored by git.
  17. Print and return. The test line prints the best epoch, accuracy, macro-F1, minutes and seconds per epoch. The function returns the row plus best_epoch, val_accuracy, seconds_per_epoch and the history dictionary.

What is written to disk: - results/metrics.csv: one row changed or added (accuracy and macro-F1 are decimals, for example 0.9827). - models/mobilenet_v2_full_frac1_seed0.pth: the best weights only. - The torch cache (the first time only): the ImageNet weights downloaded by torchvision. - Nothing else. The per-epoch history is returned in memory, not written to a file. The best epoch is printed and returned, but not saved in the CSV.

6. Why these choices

  • One shared train(). Every model is trained with the same loop, the same optimiser setup, the same split, the same scoring and the same row format. Differences in the results then come from the models, not from the code. This is the brief's main requirement for Part D.
  • A fixed seed. The seed makes the starting weights, the batch order, the augmentation and the subset repeatable. A teammate can run the same command and get the same numbers (up to tiny differences between machines). The seed is also the way to compare two settings fairly.
  • Several seeds (0, 1, 2). One run can be lucky or unlucky. With three seeds we can show the mean and the spread. Seeds 0 to 2 were run on Nitin's GPU laptop. Seed 3 is still missing, and seed 4 exists only for three scratch-CNN settings (from Rohan's CPU script).
  • Validation chooses the epoch. The validation set is a practice exam for decisions. Choosing the best epoch on validation avoids tuning the model to the test set. The test score stays an honest estimate.
  • Test once. Scoring the test set after training, once per run and never for a decision, means its numbers are not inflated.
  • Macro-F1 as well as accuracy. The classes have different sizes. Macro-F1 gives each class the same weight, so a weak small class cannot hide behind a good accuracy.
  • Keep the data in memory. Decoding JPEGs in every epoch was too slow (about 45 to 90 seconds per epoch on a CPU laptop). In memory, the same maths runs in about 4.6 seconds per epoch on the RTX 3050. We checked that the pixels match the original loader.
  • Augmentation on the GPU. Rotations and flips are exact for satellite tiles. Doing them on the GPU in batches is fast.
  • A log x-axis. The training sizes grow by about two times at each step (945, 1,890, 4,725, 9,450, 18,900). On a log axis the steps are evenly spaced, and the small sizes are not squeezed together.
  • Mean ± sample SD with ddof=1. ddof=1 is the correct SD for a small sample of seeds. The error bars show how much the seeds disagree.
  • Replace rows instead of adding them. A rerun of the same run must not duplicate the row. This keeps the table clean and makes the notebook safe to restart.
  • Only the full-data weights are saved. The website and the evaluation need full-data models. Saving every small run would fill the disk.
  • Lower learning rate for full fine-tuning. Pretrained weights are already good, so a big step could damage them. 1e-4 changes them gently.

Points to be careful about in the viva:

  • The 100% scratch CNN rows came from Tejash's own notebook helper, not from train(). According to the log, train() reproduced his mean (92.79% against 92.72%) with seeds 0 to 2.
  • Frozen transfer models still update BatchNorm averages (see the list above).
  • All sizes get the same budget of up to 20 epochs. Small sets get fewer updates and may be under-trained.
  • The study uses only three seeds for the main table. Seed 4 rows exist for the scratch CNN at 5%, 10% and 25% (from Rohan's own CPU script, not train()), but they are not in the study table.
  • Grad-CAM on 64 x 64 tiles is coarse. The control experiment shows that hiding the hottest and the coldest quarter cost almost the same confidence (0.064 and 0.061), so the heat map is a picture, not proof.
